//@version=6
// Modal $1000, Trade Size $100, Margin 5% = Leverage 20x Bawaan
strategy("Smart Trail - Bulletproof V3", overlay=true, initial_capital=1000, default_qty_type=strategy.cash, default_qty_value=100, margin_long=5, margin_short=5, commission_type=strategy.commission.percent, commission_value=0.05)

import ayvaliktrading/EyopsTelegram/1 as LAF

// ========================================================================= //
// 1. SMART TRAIL SETTINGS
// ========================================================================= //
groupST = "⚙️ SMART TRAIL SETTINGS"
st_val = input.int(4, "Sensitivity", options=[1,2,3,4,5], group=groupST)

// ========================================================================= //
// 2. RISK MANAGEMENT
// ========================================================================= //
groupRisk = "⚖️ RISK MANAGEMENT"
riskMode = input.string("Fixed %", "Tipe TP & SL", options=["Fixed %", "ATR Dinamis"], group=groupRisk)
tp_perc = input.float(4.0, "Take Profit (%)", step=0.1, group=groupRisk)
sl_perc = input.float(0.6, "Stop Loss (%)", step=0.1, group=groupRisk)
atr_len = input.int(14, "ATR Period", group=groupRisk)
tp_atr_mult = input.float(2.0, "TP ATR Multiplier", step=0.1, group=groupRisk)
sl_atr_mult = input.float(1.5, "SL ATR Multiplier", step=0.1, group=groupRisk)

// ========================================================================= //
// 3. TRAILING STOP
// ========================================================================= //
groupTrail = "🏃‍♂️ TRAILING STOP"
useTrailing = input.bool(true, "Aktifkan Trailing Stop", group=groupTrail)
trailTrigger = input.float(1.0, "Trigger P/L (%)", step=0.1, group=groupTrail)
trailDist = input.float(0.3, "Trailing Distance (%)", step=0.1, group=groupTrail)

// ========================================================================= //
// 4. AUTO-SLEEP (MESIN ANTI-ERROR V3)
// ========================================================================= //
groupSleep = "⏳ AUTO-SLEEP (CIRCUIT BREAKER)"
useSleep = input.bool(true, "Aktifkan Auto-Sleep", group=groupSleep)
maxLoss = input.int(3, "Maksimal Loss Beruntun", group=groupSleep)
sleepHours = input.float(10.0, "Lama Tidur (Jam)", group=groupSleep)

var int loss_streak = 0
var float cooldown_time = 0.0

// Deteksi otomatis tanpa ta.change (100% Kebal Error)
if strategy.losstrades > nz(strategy.losstrades[1])
    loss_streak += 1
if strategy.wintrades > nz(strategy.wintrades[1])
    loss_streak := 0

if loss_streak >= maxLoss and useSleep
    cooldown_time := time + (sleepHours * 60 * 60 * 1000)
    loss_streak := 0 

bool isSleeping = time < cooldown_time
bgcolor(isSleeping ? color.new(color.purple, 85) : na, title="Sleep Mode Background")

// ========================================================================= //
// 5. INDIKATOR UTAMA
// ========================================================================= //
atr_val = ta.atr(atr_len)
var float entry_atr = na

[smartTrailLine, fillerLine, smartTrailDirection] = LAF.getSmartTrail(10, st_val, 8) 
color_st = smartTrailDirection == 'long' ? #2157f9 : smartTrailDirection == 'short' ? #ff1100 : na
plot(smartTrailLine, "Smart Trail", color=color_st, linewidth=2, editable=false)

bool isLong = smartTrailDirection == 'long' and smartTrailDirection[1] != 'long'
bool isShort = smartTrailDirection == 'short' and smartTrailDirection[1] != 'short'

// ========================================================================= //
// 6. MEMORI TRAILING STOP
// ========================================================================= //
var bool trail_active_long = false
var float trail_stop_long = na
var bool trail_active_short = false
var float trail_stop_short = na

if isLong or isShort
    trail_active_long := false
    trail_stop_long := na
    trail_active_short := false
    trail_stop_short := na

// ========================================================================= //
// 7. EKSEKUSI PASAR (TV MENGHITUNG LOT OTOMATIS)
// ========================================================================= //
if isLong and not isSleeping
    entry_atr := atr_val
    strategy.entry("Beli (Long)", strategy.long) 

if isShort and not isSleeping
    entry_atr := atr_val
    strategy.entry("Jual (Short)", strategy.short)

// ========================================================================= //
// 8. EXIT LOGIC & HUD VISUAL
// ========================================================================= //
float avg_price = strategy.position_avg_price
float final_sl_long = na
float final_tp_long = na
float final_sl_short = na
float final_tp_short = na

if strategy.position_size > 0 
    if riskMode == "Fixed %"
        final_sl_long := avg_price * (1 - (sl_perc / 100))
        final_tp_long := avg_price * (1 + (tp_perc / 100))
    else
        final_sl_long := avg_price - (entry_atr * sl_atr_mult)
        final_tp_long := avg_price + (entry_atr * tp_atr_mult)

    if useTrailing
        float trigger_price = avg_price * (1 + (trailTrigger / 100))
        if high >= trigger_price
            trail_active_long := true
            
        if trail_active_long
            float current_trail = high * (1 - (trailDist / 100))
            trail_stop_long := math.max(nz(trail_stop_long, current_trail), current_trail)
            
        final_sl_long := trail_active_long ? trail_stop_long : final_sl_long

    strategy.exit("Tutup Long", from_entry="Beli (Long)", limit=final_tp_long, stop=final_sl_long)

if strategy.position_size < 0 
    if riskMode == "Fixed %"
        final_sl_short := avg_price * (1 + (sl_perc / 100))
        final_tp_short := avg_price * (1 - (tp_perc / 100))
    else
        final_sl_short := avg_price + (entry_atr * sl_atr_mult)
        final_tp_short := avg_price - (entry_atr * tp_atr_mult)

    if useTrailing
        float trigger_price_sh = avg_price * (1 - (trailTrigger / 100))
        if low <= trigger_price_sh
            trail_active_short := true
            
        if trail_active_short
            float current_trail_sh = low * (1 + (trailDist / 100))
            trail_stop_short := math.min(nz(trail_stop_short, current_trail_sh), current_trail_sh)
            
        final_sl_short := trail_active_short ? trail_stop_short : final_sl_short

    strategy.exit("Tutup Short", from_entry="Jual (Short)", limit=final_tp_short, stop=final_sl_short)

plot(strategy.position_size > 0 ? final_tp_long : strategy.position_size < 0 ? final_tp_short : na, title="Target TP", color=#00e676, style=plot.style_linebr, linewidth=2)
plot(strategy.position_size > 0 ? final_sl_long : strategy.position_size < 0 ? final_sl_short : na, title="Batas SL / Trailing SL", color=#ff1744, style=plot.style_linebr, linewidth=2)
